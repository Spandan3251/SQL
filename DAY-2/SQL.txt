CREATE TABLE CLIENT_MASTER53(
	CLIENT_NO VARCHAR2(6) PRIMARY KEY,
	NAME VARCHAR2(20) NOT NULL,
	CITY VARCHAR2(15),
	PINCODE NUMBER(8),
	STATE VARCHAR2(15),
	BAL_DUE NUMBER(10,2),
	CHECK(CLIENT_NO LIKE 'C%')
); 

CREATE TABLE PRODUCT_MASTER53(
	PRODUCT_NO VARCHAR2(6) PRIMARY KEY,
	DESCRIPTION VARCHAR2(15) NOT NULL,
	PROFIT_PERCENT NUMBER(4,2) NOT NULL,
	UNIT_MEASURE VARCHAR2(10) NOT NULL,
	QTY_ON_HAND NUMBER(8) NOT NULL,
	REORDER_LVL NUMBER(8) NOT NULL,
	SELL_PRICE NUMBER(8,2) NOT NULL CHECK(SELL_PRICE>0),
	COST_PRICE NUMBER(8,2) NOT NULL CHECK(COST_PRICE>0),
	CHECK(PRODUCT_NO LIKE 'P%')
);

CREATE TABLE SALESMAN_MASTER53(
	SALESMAN_NO VARCHAR2(6) PRIMARY KEY,
	SALESMAN_NAME VARCHAR2(20) NOT NULL,
	ADDRESS1 VARCHAR2(10) NOT NULL,
	ADDRESS2 VARCHAR2(10),
	CITY VARCHAR2(20),
	PINCODE NUMBER(7),
	STATE VARCHAR2(20),
	SAL_AMT NUMBER(8,2) NOT NULL CHECK(SAL_AMT >0),
	TGT_TO_GET NUMBER(6,2) NOT NULL CHECK(TGT_TO_GET >0),
	YTD_SALES NUMBER(6,2) NOT NULL,
	REMARKS VARCHAR2(20),
	CHECK(SALESMAN_NO LIKE 'S%')
);

CREATE TABLE SALES_ORDER53(
	ORDER_NO VARCHAR2(6) PRIMARY KEY,
	ORDER_DATE DATE,
	CLIENT_NO VARCHAR2(6) REFERENCES CLIENT_MASTER53(CLIENT_NO),
	DELY_ADDRESS VARCHAR2(25),
	SALESMAN_NO VARCHAR(6) REFERENCES SALESMAN_MASTER53(SALESMAN_NO),
	DELY_TYPE CHAR(1) DEFAULT 'F' CHECK (DELY_TYPE IN ('P','F')),
	BILLED_YN CHAR(1),
	DELY_DATE DATE,
	ORDER_STATUS VARCHAR2(10) CHECK (ORDER_STATUS IN ('IN PROCESS','FULFILLED','BACKORDER','CANCELLED')),
	CHECK(ORDER_NO LIKE 'O%'),
	CHECK(DELY_DATE>ORDER_DATE)
);

CREATE TABLE SALES_ORDER_DETAILS53 (
    ORDER_NO VARCHAR2(6),
    PRODUCT_NO VARCHAR2(6),
    QTY_ORDERED NUMBER(8),
    QTY_DISP NUMBER(8),
    PRODUCT_RATE NUMBER(10,2),
    PRIMARY KEY (ORDER_NO, PRODUCT_NO),
    FOREIGN KEY (ORDER_NO) REFERENCES SALES_ORDER53(ORDER_NO),
    FOREIGN KEY (PRODUCT_NO) REFERENCES PRODUCT_MASTER53(PRODUCT_NO)
);
